# **4. Identity Service**

## `identity/app.py`

```python id="identity_service"
from fastapi import FastAPI
import redis, uuid, json

app = FastAPI()
r = redis.Redis(host="redis", port=6379)

@app.post("/auth")
def auth(user: dict):
    ct = str(uuid.uuid4())

    event = {
        "type": "AUTH",
        "user": user["id"],
        "ct": ct
    }

    r.xadd("asta-events", event)
    return event
```

# **5. FSM Engine (Core ASTA-TP State Machine)**

## `fsm/app.py`

```python id="fsm_engine"
from fastapi import FastAPI
import redis, json

app = FastAPI()
r = redis.Redis(host="redis", port=6379)

STATE = {}

def publish(event):
    r.xadd("asta-events", event)

@app.post("/deauth")
def deauth(user: dict):
    event = {
        "type": "DEAUTH",
        "user": user["id"]
    }
    STATE[user["id"]] = "DEAUTH"
    publish(event)
    return event

@app.post("/reauth")
def reauth(user: dict):
    event = {
        "type": "REAUTH",
        "user": user["id"]
    }
    STATE[user["id"]] = "AUTH"
    publish(event)
    return event
```

# **6. Telepresence Orchestrator (Iterable Pods)**

## `telepresence/app.py`

```python id="telepresence"
from fastapi import FastAPI
import redis, json

app = FastAPI()
r = redis.Redis(host="redis", port=6379)

ACTIVE = {}

def listen():
    pubsub = r.pubsub()
    pubsub.subscribe("asta-events")

    for msg in pubsub.listen():
        if msg["type"] != "message":
            continue

        event = msg["data"].decode()

        if "DEAUTH" in event:
            print("KILLING TELEPRESENCE SESSION")

        if "AUTH" in event:
            print("STARTING TELEPRESENCE SESSION")

@app.on_event("startup")
def startup():
    import threading
    t = threading.Thread(target=listen)
    t.start()
```

# **7. SOC Dashboard Backend**

## `dashboard/server.py`

```python id="dashboard_backend"
from fastapi import FastAPI, WebSocket
import redis

app = FastAPI()
r = redis.Redis(host="redis", port=6379)

@app.websocket("/ws")
async def ws(websocket: WebSocket):
    await websocket.accept()

    pubsub = r.pubsub()
    pubsub.subscribe("asta-events")

    for msg in pubsub.listen():
        if msg["type"] == "message":
            await websocket.send_text(msg["data"].decode())
```

# **9. Attack Injection Engine**

This simulates real-world ASTA-TP threat classes:

## `attack_sim/app.py`

```python id="attack_sim"
import redis, time, random

r = redis.Redis(host="redis", port=6379)

users = ["user-1", "user-2"]

while True:
    u = random.choice(users)

    # simulate deauth attack
    r.xadd("asta-events", {
        "type": "DEAUTH",
        "user": u
    })

    time.sleep(5)

    # simulate spoof attempt
    r.xadd("asta-events", {
        "type": "AUTH",
        "user": u
    })

    time.sleep(10)
```
